cmake_minimum_required(VERSION 3.25)
project(pqcoin VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(PQCOIN_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
find_package(OpenSSL 3.5 REQUIRED COMPONENTS Crypto SSL)
add_library(pqcoin_crypto src/crypto.cpp)
target_include_directories(pqcoin_crypto PUBLIC include)
target_link_libraries(pqcoin_crypto PUBLIC OpenSSL::Crypto)
target_compile_options(pqcoin_crypto PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
if(PQCOIN_SANITIZERS)
  target_compile_options(pqcoin_crypto PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(pqcoin_crypto PUBLIC -fsanitize=address,undefined)
endif()
add_library(pqcoin_ledger src/ledger.cpp)
target_link_libraries(pqcoin_ledger PUBLIC pqcoin_crypto)
target_compile_options(pqcoin_ledger PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
add_library(pqcoin_app src/app.cpp src/app_bridge.cpp)
target_link_libraries(pqcoin_app PUBLIC pqcoin_ledger)
target_compile_options(pqcoin_app PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
add_library(pqcoin_pow src/pow.cpp)
target_link_libraries(pqcoin_pow PUBLIC pqcoin_app OpenSSL::SSL)
target_compile_options(pqcoin_pow PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
add_library(pqcoin_genesis src/genesis.cpp)
target_link_libraries(pqcoin_genesis PUBLIC pqcoin_crypto)
target_compile_options(pqcoin_genesis PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
add_executable(pqcoin-genesis src/genesis_main.cpp)
target_link_libraries(pqcoin-genesis PRIVATE pqcoin_genesis)
add_executable(qcash src/pow_main.cpp)
target_link_libraries(qcash PRIVATE pqcoin_pow pqcoin_genesis)
target_compile_options(qcash PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
add_executable(pqcoin-pow src/pow_main.cpp)
target_link_libraries(pqcoin-pow PRIVATE pqcoin_pow pqcoin_genesis)
target_compile_options(pqcoin-pow PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
add_executable(pqcoin-demo src/main.cpp)
target_link_libraries(pqcoin-demo PRIVATE pqcoin_ledger)

option(PQCOIN_FUZZ "Build Clang libFuzzer parser harness" OFF)
if(PQCOIN_FUZZ)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "PQCOIN_FUZZ requires Clang")
  endif()
  foreach(target pqcoin_crypto pqcoin_ledger pqcoin_app pqcoin_pow)
    target_compile_options(${target} PRIVATE -fsanitize=fuzzer-no-link,address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} INTERFACE -fsanitize=fuzzer-no-link,address,undefined)
  endforeach()
  add_executable(pqcoin-parser-fuzz tests/parser_fuzz.cpp)
  target_link_libraries(pqcoin-parser-fuzz PRIVATE pqcoin_pow)
  target_compile_options(pqcoin-parser-fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_options(pqcoin-parser-fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
endif()

option(QCASH_VULKAN "Build the optional native Vulkan mining backend" OFF)
if(QCASH_VULKAN)
  find_package(Vulkan REQUIRED)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  find_program(GLSLANG_VALIDATOR glslangValidator REQUIRED)
  set(MINING_HEADERS)
  foreach(variant scalar64 pair32 compact64 compact32 row32 unroll32 final32 tree32 lean32 column32 tail32 early32 prepare)
    set(DEFINES)
    if(variant MATCHES "32$")
      list(APPEND DEFINES -DPAIR32=1)
    endif()
    if(variant MATCHES "^compact")
      list(APPEND DEFINES -DCOMPACT=1)
    endif()
    if(variant STREQUAL "row32")
      list(APPEND DEFINES -DROW_FUSED=1)
    endif()
    if(variant STREQUAL "unroll32")
      list(APPEND DEFINES -DUNROLL_PAIRS=1)
    endif()
    if(variant MATCHES "^(final|tree|lean|column|tail|early)32$")
      list(APPEND DEFINES -DPRUNE_FINAL=1)
    endif()
    if(variant STREQUAL "tree32")
      list(APPEND DEFINES -DBALANCED_PARITY=1)
    endif()
    if(variant STREQUAL "lean32")
      list(APPEND DEFINES -DCOMPACT=1)
    endif()
    if(variant STREQUAL "column32")
      list(APPEND DEFINES -DCOLUMN_SCHEDULE=1)
    endif()
    if(variant STREQUAL "tail32")
      list(APPEND DEFINES -DROLLING_FINAL=1)
    endif()
    if(variant STREQUAL "early32")
      list(APPEND DEFINES -DEARLY_TARGET=1)
    endif()
    set(SHADER mining.comp)
    if(variant STREQUAL "prepare")
      set(SHADER prepare.comp)
    endif()
    add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/mining_${variant}.spv
      COMMAND ${GLSLANG_VALIDATOR} -V --target-env vulkan1.1 -Os ${DEFINES} ${CMAKE_CURRENT_SOURCE_DIR}/shaders/${SHADER} -o ${CMAKE_CURRENT_BINARY_DIR}/mining_${variant}.spv
      DEPENDS shaders/${SHADER} shaders/keccak.glsl VERBATIM)
    add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/mining_${variant}.hpp
      COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/embed_spirv.py ${CMAKE_CURRENT_BINARY_DIR}/mining_${variant}.spv ${CMAKE_CURRENT_BINARY_DIR}/mining_${variant}.hpp mining_${variant}
      DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/mining_${variant}.spv tools/embed_spirv.py VERBATIM)
    list(APPEND MINING_HEADERS ${CMAKE_CURRENT_BINARY_DIR}/mining_${variant}.hpp)
  endforeach()
  add_library(qcash_vulkan src/vulkan_miner.cpp ${MINING_HEADERS})
  target_include_directories(qcash_vulkan PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
  target_link_libraries(qcash_vulkan PUBLIC pqcoin_crypto Vulkan::Vulkan)
  target_compile_options(qcash_vulkan PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wno-missing-field-initializers)
  add_executable(qcash-vulkan src/vulkan_main.cpp)
  target_link_libraries(qcash-vulkan PRIVATE qcash_vulkan pqcoin_pow)
  if(BUILD_TESTING)
    add_executable(qcash-vulkan-tests tests/vulkan_test.cpp)
    target_link_libraries(qcash-vulkan-tests PRIVATE qcash_vulkan pqcoin_pow)
    add_test(NAME vulkan-hash-search COMMAND qcash-vulkan-tests)
    set_tests_properties(vulkan-hash-search PROPERTIES TIMEOUT 120)
    add_test(NAME vulkan-mining-worker COMMAND ${CMAKE_COMMAND} -E env
      QCASH_BINARY=$<TARGET_FILE:qcash> QCASH_VULKAN_BINARY=$<TARGET_FILE:qcash-vulkan>
      ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/vulkan_worker_test.py)
    set_tests_properties(vulkan-mining-worker PROPERTIES TIMEOUT 120)
    add_test(NAME vulkan-counter-reader COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/vulkan_counter_reader_test.py)
    set_tests_properties(vulkan-counter-reader PROPERTIES TIMEOUT 30)
  endif()
endif()
