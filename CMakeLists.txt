cmake_minimum_required(VERSION 3.25)
project(pqcoin VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(PQCOIN_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
find_package(OpenSSL 3.5 REQUIRED COMPONENTS Crypto SSL)
add_library(pqcoin_crypto src/crypto.cpp)
target_include_directories(pqcoin_crypto PUBLIC include)
target_link_libraries(pqcoin_crypto PUBLIC OpenSSL::Crypto)
target_compile_options(pqcoin_crypto PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
if(PQCOIN_SANITIZERS)
  target_compile_options(pqcoin_crypto PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(pqcoin_crypto PUBLIC -fsanitize=address,undefined)
endif()
add_library(pqcoin_ledger src/ledger.cpp)
target_link_libraries(pqcoin_ledger PUBLIC pqcoin_crypto)
target_compile_options(pqcoin_ledger PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
add_library(pqcoin_app src/app.cpp src/app_bridge.cpp)
target_link_libraries(pqcoin_app PUBLIC pqcoin_ledger)
target_compile_options(pqcoin_app PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
add_library(pqcoin_pow src/pow.cpp)
target_link_libraries(pqcoin_pow PUBLIC pqcoin_app OpenSSL::SSL)
target_compile_options(pqcoin_pow PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
add_library(pqcoin_genesis src/genesis.cpp)
target_link_libraries(pqcoin_genesis PUBLIC pqcoin_crypto)
target_compile_options(pqcoin_genesis PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
add_executable(pqcoin-genesis src/genesis_main.cpp)
target_link_libraries(pqcoin-genesis PRIVATE pqcoin_genesis)
add_executable(qcash src/pow_main.cpp)
target_link_libraries(qcash PRIVATE pqcoin_pow pqcoin_genesis)
target_compile_options(qcash PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
add_executable(pqcoin-pow src/pow_main.cpp)
target_link_libraries(pqcoin-pow PRIVATE pqcoin_pow pqcoin_genesis)
target_compile_options(pqcoin-pow PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
add_executable(pqcoin-demo src/main.cpp)
target_link_libraries(pqcoin-demo PRIVATE pqcoin_ledger)
include(CTest)
if(BUILD_TESTING)
  add_executable(pqcoin-consensus-fixture tests/consensus_fixture.cpp)
  target_link_libraries(pqcoin-consensus-fixture PRIVATE pqcoin_app pqcoin_genesis)
  add_executable(pqcoin-tests tests/crypto_test.cpp)
  target_link_libraries(pqcoin-tests PRIVATE pqcoin_ledger)
  add_test(NAME crypto-transfer COMMAND pqcoin-tests)
  add_executable(pqcoin-genesis-tests tests/genesis_test.cpp)
  target_link_libraries(pqcoin-genesis-tests PRIVATE pqcoin_genesis)
  add_test(NAME genesis COMMAND pqcoin-genesis-tests)
  add_executable(pqcoin-app-tests tests/app_test.cpp)
  target_link_libraries(pqcoin-app-tests PRIVATE pqcoin_app pqcoin_genesis)
  add_test(NAME application COMMAND pqcoin-app-tests)
  add_executable(pqcoin-pow-tests tests/pow_test.cpp)
  target_link_libraries(pqcoin-pow-tests PRIVATE pqcoin_pow)
  add_test(NAME proof-of-work COMMAND pqcoin-pow-tests)
  add_executable(pqcoin-consensus-adversary tests/consensus_adversary.cpp)
  target_link_libraries(pqcoin-consensus-adversary PRIVATE pqcoin_pow)
  add_test(NAME consensus-adversary COMMAND pqcoin-consensus-adversary)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME quantum-circuit-model COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/circuit_model_test.py)
  add_test(NAME quantum-resource-model COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/quantum_model_test.py)
endif()

option(PQCOIN_FUZZ "Build Clang libFuzzer parser harness" OFF)
if(PQCOIN_FUZZ)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "PQCOIN_FUZZ requires Clang")
  endif()
  foreach(target pqcoin_crypto pqcoin_ledger pqcoin_app pqcoin_pow)
    target_compile_options(${target} PRIVATE -fsanitize=fuzzer-no-link,address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} INTERFACE -fsanitize=fuzzer-no-link,address,undefined)
  endforeach()
  add_executable(pqcoin-parser-fuzz tests/parser_fuzz.cpp)
  target_link_libraries(pqcoin-parser-fuzz PRIVATE pqcoin_pow)
  target_compile_options(pqcoin-parser-fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_options(pqcoin-parser-fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
